# MediaForge Python dependencies

# ─── 列在這裡的判準 ───

# ─── FFmpeg / FFprobe binaries (雙保險) ───
# 系統有裝 ffmpeg 是優先 (最快、最相容、codec 最齊)，沒裝的話這兩個 fallback 撐住：
#   - imageio-ffmpeg: 純 ffmpeg static binary (~70MB)
#   - static-ffmpeg:  ffmpeg + ffprobe binaries，lazy download on first use (~30-60MB)
# 解析順序 (utils/ffmpeg.py:get_ffmpeg_bin / get_ffprobe_bin)：
#   ffmpeg :  system → imageio-ffmpeg → static-ffmpeg
#   ffprobe:  system → static-ffmpeg (imageio-ffmpeg 不帶 ffprobe)
imageio-ffmpeg>=0.4.9
static-ffmpeg>=2.5

# ─── 中文轉換 ───
# MF_ConvertChinese 唯一 backend，pure Python 零 C++ 依賴
opencc-python-reimplemented>=0.1.7

# ─── HTTP / 字型解析 (輕量、純 Python) ───
# MF_WhisperTranscribe (openai_compatible) / MF_TranslateSubtitle
requests>=2.28
# MF_BurnSubtitle 自動讀 TTF Family Name；沒裝會 fallback 到檔名 stem
fontTools>=4.0
